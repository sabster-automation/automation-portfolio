Section 1

Overview — what this app is and isn't

What it is: A production-grade Playwright + TypeScript automation portfolio that tests two real applications under test (AUTs) — SauceDemo (e-commerce checkout) and DemoQA Practice Form (form with file upload, date picker, react-select, modal) — plus API (jsonplaceholder / reqres), visual regression, accessibility (axe-core), Allure Reporting, and multi-environment execution.

What it is not: Not a product backend. There is no app server to run — the "back-end" in this repo is the test-side API layer (utils/apiClient.ts, utils/seedHelper.ts, config/environments.ts) that talks to external public APIs per environment. There is no database to migrate.

Request Flow (per test)

  1. ENV resolved in playwright.config.ts:6-8 → getEnvConfig() / getBaseURL() → baseURL injected into use.
  2. Fixture creates POMs (lazy, per-worker isolated). authenticatedPage logs in via LoginPage.loginWithEnvDefaults(); seededCustomer calls seedCustomerViaAPI → ApiClient.createUser → Zod parse → returns {...customer, id, seeded}; teardown → cleanupCustomer.
  3. Spec drives POM methods only (no direct locators). Assertions live in POM or spec with expect.
  4. Reports emitted per reporter: [html, junit, allure-playwright, list]; trace/screenshot/video on failure.
Key invariant: E2E projects do not depend on setup — SauceDemo keeps session in sessionStorage, so storageState is written but not reused. Each spec logs in itself via fixtures/test-fixtures.ts:35-41. Don't add dependencies:['setup'] to e2e projects.

Section 5

Prerequisites

  • Node.js 20+ (CI uses 20; 18 also works for Playwright 1.49).
  • npm (lockfile is package-lock.json — use npm ci, not npm install in CI).
  • OS: Windows / macOS / Linux. Visual snapshots are OS-specific — Linux baselines must be generated on Linux (CI workflow does this).
  • No Docker, no DB, no app server — AUTs are public URLs.

Environment config

config/environments.ts:9 — baseURL always saucedemo; apiBaseURL switches.

// config/environments.ts
export type Environment = 'dev'|'staging'|'prod';
export interface EnvConfig { baseURL:string; apiBaseURL:string; name:string; }
export const environments: Record<Environment,EnvConfig> = {
  dev:     { name:'Development', baseURL:'https://www.saucedemo.com', apiBaseURL:'https://jsonplaceholder.typicode.com' },
  staging: { name:'Staging',     baseURL:'https://www.saucedemo.com', apiBaseURL:'https://reqres.in/api' },
  prod:    { name:'Production',  baseURL:'https://www.saucedemo.com', apiBaseURL:'https://jsonplaceholder.typicode.com' },
};
export function getEnvConfig(): EnvConfig {
  const env = (process.env.ENV as Environment) || 'dev';
  if (!environments[env]) throw new Error(`Unknown ENV="${env}". Valid: ${Object.keys(environments).join(', ')}`);
  return environments[env];
}
export function getBaseURL(): string {
  if (process.env.BASE_URL) return process.env.BASE_URL; // override wins
  return getEnvConfig().baseURL;
}

Per-env behavior: dev/prod use /posts (jsonplaceholder, ephemeral — DELETE is no-op), staging uses /api/users (reqres, DELETE 204). Prod cleanup is no-op (dedicated tenant, nightly purge).

Test data factories

test-data/factories/customerFactory.ts — faker, overrides, DemoQA constraints.

export interface Customer { firstName,lastName,email,gender,mobile,dateOfBirth,subject,hobby,address,state,city,picturePath? }
export function createCustomer(overrides: Partial<Customer>={}): Customer {
  const firstName=faker.person.firstName(), lastName=faker.person.lastName();
  return {
    firstName, lastName,
    email: faker.internet.email({firstName,lastName}).toLowerCase(),
    gender: faker.helpers.arrayElement(['Male','Female','Other']),
    mobile: faker.string.numeric(10), // DemoQA needs 10 digits
    dateOfBirth: '10 Jan 1990',        // static — random dates flake on react-datepicker
    subject: faker.helpers.arrayElement(['Maths','Computer Science','Physics']),
    hobby: faker.helpers.arrayElement(['Sports','Reading','Music']),
    address: faker.location.streetAddress(),
    state: 'NCR', city: 'Delhi',       // DemoQA valid pair
    ...overrides,
  };
}
export function createCheckoutCustomer(overrides) { return { ...createCustomer(overrides), postalCode: faker.location.zipCode('#####'), ...overrides }; }

API Client & Contracts

Location: utils/apiClient.ts:10 — typed wrapper around Playwright request with Zod, per-env baseURL.

export const UserSchema = z.object({
  id: z.union([z.number(), z.string()]),
  email: z.string().email(),
  name: z.string().optional(),
});
export const CreateUserResponseSchema = z.object({
  id: z.union([z.number(), z.string()]),
  name: z.string().optional(), job: z.string().optional(),
  title: z.string().optional(), email: z.string().optional(), createdAt: z.string().optional(),
});

function getApiBase(): string { return process.env.API_BASE_URL || getEnvConfig().apiBaseURL; }

export class ApiClient {
  constructor(private request: APIRequestContext) {}
  private get base(){ return getApiBase(); }
  isReqres(): boolean { return this.base.includes('reqres'); }

  async getUsers() {
    const res = await this.request.get(`${this.base}/users`);
    expect(res.ok()).toBeTruthy();
    const body = await res.json();
    const users = Array.isArray(body) ? body : body.data; // reqres wraps in {data:[]}
    return users as User[];
  }
  async getUser(id) {
    const res = await this.request.get(`${this.base}/users/${id}`);
    expect([200,404]).toContain(res.status());
    if (res.status()===404) return null;
    const data = (await res.json()).data || await res.json(); // unwrap
    UserSchema.parse(data); // contract validation
    return data;
  }
  async createUser(payload) {
    const endpoint = this.isReqres() ? `${this.base}/users` : `${this.base}/posts`;
    const res = await this.request.post(endpoint, { data: payload });
    expect([200,201]).toContain(res.status());
    CreateUserResponseSchema.parse(await res.json());
    return body;
  }
  async deleteUser(id) {
    if (getEnvConfig().name==='Production') return; // dedicated tenant — never delete prod
    const endpoint = this.isReqres() ? `${this.base}/users/${id}` : `${this.base}/posts/${id}`;
    const res = await this.request.delete(endpoint);
    expect([200,204,404]).toContain(res.status()); // best-effort
  }
}

export async function seedCustomerViaAPI(request, overrides={}): Promise<SeededCustomer> {
  const customer = createCustomer(overrides);
  const client = new ApiClient(request);
  const payload = client.isReqres()
    ? { name: `${customer.firstName} ${customer.lastName}`, job: customer.subject, email: customer.email }
    : { title: `${customer.firstName} ${customer.lastName}`, body: customer.email, userId: 1 };
  const created = await client.createUser(payload);
  return { ...customer, id: created.id ?? `temp-${Date.now()}`, seeded: true };
}

}
